// 1. 金額を整数（最小通貨単位）に変換するユーティリティ
const toCents = (amount: number): number => Math.round(amount * 100);

// 2. スコアリング関数
const calculateConfidence = (targetAmount: number, targetTime: number, candidates: any[]): number => {
  let score = 100;
  // 日付の乖離による減点 (1日離れるごとに -5点)
  const avgTime = candidates.reduce((sum, c) => sum + new Date(c.date).getTime(), 0) / candidates.length;
  const daysDiff = Math.abs(targetTime - avgTime) / (1000 * 60 * 60 * 24);
  score -= daysDiff * 5;

  // 要素数による減点 (組み合わせが多いほど偶然の可能性が上がるため)
  score -= (candidates.length - 1) * 10;
  
  return score;
};

// 3. 枝刈り付きバックトラッキング（1対N の探索）
const findSubsetSum = (
  targetAmount: number,
  candidates: any[],
  maxDepth: number = 5
): any[][] => {
  const targetCents = toCents(targetAmount);
  // 枝刈りを効率化するため、金額の降順にソート（大きいものから試す）
  const sortedCandidates = [...candidates]
    .map(c => ({...c, cents: toCents((c.expense || 0) - (c.income || 0))}))
    .filter(c => Math.abs(c.cents) <= Math.abs(targetCents)) // 同じ極性のみ、かつターゲット金額以下
    .sort((a, b) => Math.abs(b.cents) - Math.abs(a.cents));

  const results: any[][] = [];

  const backtrack = (
    startIndex: number,
    currentSum: number,
    currentCombination: any[]
  ) => {
    // 成功条件
    if (currentSum === targetCents) {
      results.push([...currentCombination]);
      return;
    }
    // 失敗・枝刈り条件
    if (Math.abs(currentSum) > Math.abs(targetCents) || currentCombination.length >= maxDepth) {
      return;
    }

    for (let i = startIndex; i < sortedCandidates.length; i++) {
      const candidate = sortedCandidates[i];
      // Check sign (prevent mixing income and expenses in the same combination)
      if (Math.sign(candidate.cents) !== Math.sign(targetCents)) continue;

      const nextSum = currentSum + candidate.cents;

      // Look-ahead 枝刈り：次の要素を足してオーバーするならスキップ
      if (Math.abs(nextSum) > Math.abs(targetCents)) continue;

      currentCombination.push(candidate);
      backtrack(i + 1, nextSum, currentCombination);
      currentCombination.pop(); // バックトラック
    }
  };

  backtrack(0, 0, []);
  return results;
};

// 1対Nマッチングのエントリポイント
const matchOneToMany = (
  targetAmt: number,
  targetTime: number,
  unmatchedCandidates: any[],
  thresholdScore: number = 60
): { matchedIndices: number[], confidenceScore: number } | null => {
  const validCombinations = findSubsetSum(targetAmt, unmatchedCandidates, 5);

  if (validCombinations.length === 0) return null;

  let bestMatch = null;
  let highestScore = -Infinity;

  for (const combo of validCombinations) {
    const score = calculateConfidence(targetAmt, targetTime, combo);
    if (score > highestScore && score >= thresholdScore) {
      highestScore = score;
      bestMatch = combo;
    }
  }

  return bestMatch ? {
    matchedIndices: bestMatch.map(tx => tx.originalIndex !== undefined ? tx.originalIndex : tx.index),
    confidenceScore: highestScore
  } : null;
};

function deduplicateBankRecords(bankRecords: any[], reconciledRecords: any[]) {
  const usedDbIndices = new Set<number>();
  const parseDate = (dstr: string) => new Date(dstr).getTime();
  const DAY_MS = 24 * 60 * 60 * 1000;

  return bankRecords.filter(b => {
    const bAmt = (b.expense || 0) - (b.income || 0);
    const bTime = parseDate(b.date);
    
    // First pass: 1-to-1 strict exact match
    let bestMatch = -1;
    let minDiff = Infinity;
    for (let j = 0; j < reconciledRecords.length; j++) {
       if (usedDbIndices.has(j)) continue;
       const a = reconciledRecords[j];
       const aAmt = (a.expense || 0) - (a.income || 0);
       const aTime = parseDate(a.date);
       if (Math.abs(bAmt - aAmt) < 0.01) {
          const daysDiff = Math.abs(bTime - aTime) / DAY_MS;
          if (daysDiff <= 3 && daysDiff < minDiff) { // strict date limit for 1-to-1
             minDiff = daysDiff;
             bestMatch = j;
          }
       }
    }
    
    if (bestMatch !== -1) {
       usedDbIndices.add(bestMatch);
       return false;
    }

    // Second pass: 1-to-N DP Branch and Bound
    const candidates: any[] = [];
    for (let j = 0; j < reconciledRecords.length; j++) {
       if (usedDbIndices.has(j)) continue;
       const aTime = parseDate(reconciledRecords[j].date);
       if (Math.abs(bTime - aTime) / DAY_MS <= 14) {
          candidates.push({ ...reconciledRecords[j], originalIndex: j });
       }
    }
    
    const match = matchOneToMany(bAmt, bTime, candidates, 60);
    if (match) {
       match.matchedIndices.forEach((idx: number) => usedDbIndices.add(idx));
       return false;
    }
    
    return true; // Keep
  });
}

function autoReconcile(bankRecords: any[], appRecords: any[]) {
  const matchedAppIndices = new Set<number>();
  const matchedBankIndices = new Set<number>();
  const parseDate = (dstr: string) => new Date(dstr).getTime();
  const DAY_MS = 24 * 60 * 60 * 1000;
  
  // Pass 1: 1-to-1 exact matches
  for (let i = 0; i < bankRecords.length; i++) {
    if (matchedBankIndices.has(i)) continue;
    const b = bankRecords[i];
    const bAmt = (b.expense || 0) - (b.income || 0);
    const bTime = parseDate(b.date);
    
    let bestMatch = -1;
    let minDiff = Infinity;
    
    for (let j = 0; j < appRecords.length; j++) {
      if (matchedAppIndices.has(appRecords[j].originalIndex)) continue;
      const a = appRecords[j];
      const aAmt = (a.expense || 0) - (a.income || 0);
      const aTime = parseDate(a.date);
      
      if (Math.abs(bAmt - aAmt) < 0.01) {
        const daysDiff = Math.abs(bTime - aTime) / DAY_MS;
        if (daysDiff <= 3 && daysDiff < minDiff) { // strict 3 days
          minDiff = daysDiff;
          bestMatch = j;
        }
      }
    }
    if (bestMatch !== -1) {
      matchedBankIndices.add(i);
      matchedAppIndices.add(appRecords[bestMatch].originalIndex);
    }
  }

  // Pass 2: 1-to-N matching (1 Bank to N App)
  for (let i = 0; i < bankRecords.length; i++) {
    if (matchedBankIndices.has(i)) continue;
    const b = bankRecords[i];
    const bAmt = (b.expense || 0) - (b.income || 0);
    const bTime = parseDate(b.date);
    
    const candidates: any[] = [];
    for (let j = 0; j < appRecords.length; j++) {
       if (matchedAppIndices.has(appRecords[j].originalIndex)) continue;
       const aTime = parseDate(appRecords[j].date);
       if (Math.abs(bTime - aTime) / DAY_MS <= 14) {
         candidates.push(appRecords[j]);
       }
    }
    
    const match = matchOneToMany(bAmt, bTime, candidates, 60);
    if (match) {
       matchedBankIndices.add(i);
       match.matchedIndices.forEach((idx: number) => matchedAppIndices.add(idx));
    }
  }
  
  // Pass 3: N-to-1 matching (N Bank to 1 App)
  for (let j = 0; j < appRecords.length; j++) {
    const a = appRecords[j];
    if (matchedAppIndices.has(a.originalIndex)) continue;
    const aAmt = (a.expense || 0) - (a.income || 0);
    const aTime = parseDate(a.date);
    
    const candidates: any[] = [];
    for (let i = 0; i < bankRecords.length; i++) {
       if (matchedBankIndices.has(i)) continue;
       const bTime = parseDate(bankRecords[i].date);
       if (Math.abs(bTime - aTime) / DAY_MS <= 14) {
         candidates.push({ ...bankRecords[i], index: i });
       }
    }
    
    const match = matchOneToMany(aAmt, aTime, candidates, 60);
    if (match) {
       matchedAppIndices.add(a.originalIndex);
       match.matchedIndices.forEach((idx: number) => matchedBankIndices.add(idx));
    }
  }

  return { matchedBankIndices, matchedAppIndices };
}

              </label>
              <div style={{ display: 'flex', gap: '10px', marginTop: '1rem' }}>
                <button type="submit" className="action-button primary" style={{ flex: 1, background: '#10b981', border: 'none' }}>相殺して予算を復活させる</button>
                <button type="button" className="action-button secondary" onClick={() => setShowOffsetModal(false)}>キャンセル</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Reconcile Modal */}
      {showReconcileModal && (
        <div className="modal-overlay" onClick={() => setShowReconcileModal(false)}>
          <div className="modal-content" onClick={e => e.stopPropagation()}>
            <h2 className="chart-title" style={{ color: '#d97706' }}>🎒 ワンクリック一括精算</h2>
            <p style={{ color: 'var(--text-secondary)', marginBottom: '1.5rem', fontSize: '0.9rem' }}>
              隔離保留中だった総額（{formatCurrency(summary?.unsettledSandbox || 0)}）を一括精算します。<br/>
              自分の実費分は「特別体験・イベント費」として記録され、残りは回収分として総残高を復元します。
            </p>
            <form onSubmit={handleReconcileSubmit} className="form-container">
              <label>
                最終的な自分の実費総額 ($)
                <input type="number" step="0.01" value={reconcileForm.myShare} onChange={e => setReconcileForm({...reconcileForm, myShare: e.target.value})} required placeholder="例: 150.00" />
              </label>
              <label>
                友人からの回収・立替精算額 ($)
                <input type="number" step="0.01" value={reconcileForm.recovered} onChange={e => setReconcileForm({...reconcileForm, recovered: e.target.value})} required placeholder="例: 350.00" />
              </label>
              <div style={{ display: 'flex', gap: '10px', marginTop: '1rem' }}>
                <button type="submit" className="action-button primary" style={{ flex: 1, background: '#f59e0b', border: 'none' }}>精算を完了する</button>
                <button type="button" className="action-button secondary" onClick={() => setShowReconcileModal(false)}>キャンセル</button>
              </div>
            </form>
          </div>
        </div>
      )}


      {/* CSV Reconcile Modal */}
      {showCsvModal && (
        <div className="modal-overlay" style={{ zIndex: 9999 }}>
          <div className="modal-content" style={{ width: '95%', maxWidth: '1200px', height: '90vh', display: 'flex', flexDirection: 'column' }}>
            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '1rem' }}>
              <div style={{ display: 'flex', alignItems: 'center', gap: '20px' }}>
                <h2 style={{ margin: 0, color: 'var(--text-primary)' }}>CSV 比較照合エンジン</h2>
                <div style={{ display: 'flex', alignItems: 'center', gap: '10px', background: 'rgba(255,255,255,0.1)', padding: '5px 15px', borderRadius: '8px', border: '1px solid var(--border-color)' }}>
                  <span style={{ fontSize: '0.9rem', color: 'var(--text-secondary)' }}>🏦 銀行残高確認:</span>
                  <input 
                    type="number" 
                    value={bankBalanceInput} 
                    onChange={e => setBankBalanceInput(e.target.value)} 
                    placeholder="現在の銀行残高を入力"
                    style={{ width: '150px', padding: '5px 10px', borderRadius: '6px', border: '1px solid var(--glass-border)', background: 'var(--bg-color)', color: 'var(--text-primary)' }}
                  />
                  {(() => {
                    const parsedBank = parseFloat(bankBalanceInput);
                    if (isNaN(parsedBank)) return null;
                    const totalAppBalance = summary?.currentBalance || 0;
                    const diff = parsedBank - totalAppBalance;
                    if (Math.abs(diff) < 0.01) {
                      return <span style={{ color: 'var(--success-color)', fontWeight: 'bold' }}>✅ 残高が一致しました！</span>;
                    }
                    return <span style={{ color: 'var(--warning-color)' }}>差額: {formatCurrency(diff)}</span>;
                  })()}
                </div>
              </div>
              <button className="icon-button" onClick={() => setShowCsvModal(false)}>✕</button>
            </div>
            
            <div style={{ display: 'flex', gap: '20px', flex: 1, minHeight: 0 }}>
              {/* Left Panel: CSV Records */}
              <div style={{ flex: 1, display: 'flex', flexDirection: 'column', background: 'var(--surface-color)', padding: '1rem', borderRadius: '12px', border: '1px solid var(--border-color)', overflow: 'hidden' }}>
                <h3 style={{ margin: '0 0 10px 0', fontSize: '1.1rem' }}>未照合のCSVデータ (銀行)</h3>
                <div style={{ overflowY: 'auto', flex: 1 }}>
                  <table className="data-table" style={{ tableLayout: "fixed", width: "100%", wordBreak: "break-all" }}>
                    <thead>
                      <tr>
                        <th style={{ width: "40px" }}>✓</th>
                        <th style={{ width: "20%" }}>日付</th>
                        <th style={{ width: "45%" }}>内容</th>
                        <th style={{ width: "15%" }}>出金</th>
                        <th style={{ width: "15%" }}>入金</th>
                      </tr>
                    </thead>
                    <tbody>
                      {csvRecords.map((r, i) => {
                        const isSelected = selectedCsvIndices.has(i);
                        return (
                          <tr key={i} onClick={() => {
                            const newSet = new Set(selectedCsvIndices);
                            if (isSelected) newSet.delete(i);
                            else newSet.add(i);
                            setSelectedCsvIndices(newSet);
                          }} style={{ cursor: 'pointer', background: isSelected ? 'rgba(99, 102, 241, 0.1)' : '' }}>
                            <td><input type="checkbox" checked={isSelected} readOnly /></td>
                            <td>{r.date}</td>
                            <td>{r.description}</td>
                            <td style={{ color: r.expense > 0 ? 'var(--expense-color)' : '' }}>{r.expense > 0 ? formatCurrency(r.expense) : ''}</td>
                            <td style={{ color: r.income > 0 ? 'var(--income-color)' : '' }}>{r.income > 0 ? formatCurrency(r.income) : ''}</td>
                          </tr>
                        );
                      })}
                      {csvRecords.length === 0 && <tr><td colSpan={5} style={{ textAlign: 'center' }}>データがありません</td></tr>}
                    </tbody>
                  </table>
                </div>
              </div>

              {/* Right Panel: App Records */}
              <div style={{ flex: 1, display: 'flex', flexDirection: 'column', background: 'var(--surface-color)', padding: '1rem', borderRadius: '12px', border: '1px solid var(--border-color)', overflow: 'hidden' }}>
                <h3 style={{ margin: '0 0 10px 0', fontSize: '1.1rem' }}>未照合のアプリデータ</h3>
                <div style={{ overflowY: 'auto', flex: 1 }}>
                  <table className="data-table" style={{ tableLayout: "fixed", width: "100%", wordBreak: "break-all" }}>
                    <thead>
                      <tr>
                        <th style={{ width: "40px" }}>✓</th>
                        <th style={{ width: "20%" }}>日付</th>
                        <th style={{ width: "25%" }}>カテゴリ</th>
                        <th style={{ width: "45%" }}>内容</th>
                        <th style={{ width: "20%" }}>金額</th>
                      </tr>
                    </thead>
                    <tbody>
                      {data?.records && data.records.map((r: any, originalIndex: number) => ({...r, originalIndex}))
                        .filter((r: any) => !r.reconciled && (r.expense > 0 || r.income > 0))
                        .map((r: any) => {
                        const isSelected = selectedAppIndices.has(r.originalIndex);
                        const isIncome = r.income > 0;
                        const amount = isIncome ? r.income : r.expense;
                        return (
                          <tr key={r.originalIndex} onClick={() => {
                            const newSet = new Set(selectedAppIndices);
                            if (isSelected) newSet.delete(r.originalIndex);
                            else newSet.add(r.originalIndex);
                            setSelectedAppIndices(newSet);
                          }} style={{ cursor: 'pointer', background: isSelected ? 'rgba(99, 102, 241, 0.1)' : '' }}>
                            <td><input type="checkbox" checked={isSelected} readOnly /></td>
                            <td>{r.date}</td>
                            <td>{r.category}</td>
                            <td>{r.description}</td>
                            <td style={{ color: isIncome ? 'var(--income-color)' : 'var(--expense-color)' }}>{formatCurrency(amount)}</td>
                          </tr>
                        );
                      })}
                    </tbody>
                  </table>
                </div>
              </div>
            </div>

            {/* Bottom Panel: Actions */}
            <div style={{ marginTop: '20px', padding: '1rem', background: 'var(--surface-color)', borderRadius: '12px', border: '1px solid var(--border-color)', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
              {(() => {
                let csvTotal = 0;
                let appTotal = 0;
                
                selectedCsvIndices.forEach(i => {
                  const r = csvRecords[i];
                  csvTotal += (r.expense || 0) - (r.income || 0);
                });
                
                selectedAppIndices.forEach(i => {
                  if(!data || !data.records) return;
                  const r = data.records[i];
                  if(r) { appTotal += (r.expense || 0) - (r.income || 0); }
                });
                
                const diff = csvTotal - appTotal;
                
                const handleReconcile = async (isAdjustment = false) => {
                  if (selectedCsvIndices.size === 0 && selectedAppIndices.size === 0) return;
                  
                  let newRecords: any[] = [];
                  if (isAdjustment && diff !== 0) {
                     const cat = window.prompt("差額分のカテゴリ名を入力してください (例: その他, 手数料, 照合調整金)", "その他");
                     if (!cat) return;
                     const today = new Date();
                     const todayStr = today.toISOString().split('T')[0].replace(/-/g, '/');
                     const isExp = diff > 0;
                     newRecords.push({
                       date: todayStr,
                       category: cat,
                       description: 'CSV照合調整',
                       expense: isExp ? diff : 0,
                       income: isExp ? 0 : -diff,
                       balance: 0,
                       month: todayStr.substring(0, 7).replace('/', '-'),
                       recordType: 'expense_normal'
                     });
                  } else if (appTotal === 0 && selectedCsvIndices.size > 0) {
                     let cat = '相殺/キャンセル';
                     if (csvTotal !== 0) {
                       const inputCat = window.prompt("CSVから新規登録するカテゴリ名を入力してください", "未分類");
                       if (!inputCat) return;
                       cat = inputCat;
                     }
                     selectedCsvIndices.forEach(i => {
                       const r = csvRecords[i];
                       newRecords.push({
                         date: r.date,
                         category: cat,
                         description: r.description,
                         expense: r.expense,
                         income: r.income,
                         balance: 0,
                         month: r.date ? r.date.substring(0, 7).replace('/', '-') : '',
                         recordType: r.expense > 0 ? 'expense_normal' : 'income_normal'
                       });
                     });
                  }

                  try {
                    const res = await fetch('/api/finance', {
                      method: 'POST',
                      headers: { 'Content-Type': 'application/json' },
                      body: JSON.stringify({
                        action: 'batch_reconcile',
                        payload: {
                          reconciledIds: Array.from(selectedAppIndices),
                          newRecords: newRecords
                        }
                      })
                    });
                    if (res.ok) {
                       alert('照合が完了しました');
                       const remainingCsv = csvRecords.filter((_, idx) => !selectedCsvIndices.has(idx));
                       setCsvRecords(remainingCsv);
                       setSelectedCsvIndices(new Set());
                       setSelectedAppIndices(new Set());
                       fetchData();
                    }
                  } catch (e) { alert('エラーが発生しました'); }
                };
                
                return (
                  <>
                    <div style={{ display: 'flex', gap: '20px', fontSize: '1.1rem', fontWeight: 'bold' }}>
                      <span style={{ color: 'var(--text-secondary)' }}>CSV選択計: <span style={{ color: csvTotal >= 0 ? 'var(--expense-color)' : 'var(--income-color)' }}>{formatCurrency(Math.abs(csvTotal))}</span> ({csvTotal >= 0 ? '出金' : '入金'})</span>
                      <span style={{ color: 'var(--text-secondary)' }}>アプリ選択計: <span style={{ color: appTotal >= 0 ? 'var(--expense-color)' : 'var(--income-color)' }}>{formatCurrency(Math.abs(appTotal))}</span></span>
                      <span style={{ color: diff === 0 ? 'var(--success-color)' : 'var(--expense-color)' }}>
                        差額: {formatCurrency(Math.abs(diff))}
                      </span>
                    </div>
                    
                    <div style={{ display: 'flex', gap: '10px' }}>
                      {diff === 0 && (selectedCsvIndices.size > 0 || selectedAppIndices.size > 0) && (
                        <button className="action-button primary" onClick={() => handleReconcile(false)}>✅ 照合確定</button>
                      )}
                      {diff !== 0 && appTotal === 0 && selectedCsvIndices.size > 0 && (
                        <button className="action-button" style={{ background: '#3b82f6', border: 'none', color: '#fff' }} onClick={() => handleReconcile(false)}>📥 CSVから新規登録</button>
                      )}
                      {diff !== 0 && (appTotal !== 0 || (appTotal === 0 && selectedCsvIndices.size === 0)) && (selectedCsvIndices.size > 0 || selectedAppIndices.size > 0) && (
                        <button className="action-button secondary" onClick={() => handleReconcile(true)}>⚖️ 差額を自動作成して照合</button>
                      )}
                    </div>
                  </>
                );
              })()}
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
